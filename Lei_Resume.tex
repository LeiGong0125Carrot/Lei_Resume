\section{SELECT PROJECTS}

\begin{twocolentry}{
    Jan 2024 -- May 2025
}
    \textbf{Boosting Clinical Outcome Prediction with Context-Aware Feature Imputation and Disentanglement}
\end{twocolentry}

\vspace{0.05 cm}
\begin{onecolentry}
    \begin{highlights}
        \item Developed a \textbf{context-aware feature imputation} framework that improves clinical outcome prediction under incomplete EHR feature settings by exploiting structural signals from semi-structured clinical narratives.
        
        \item Designed a \textbf{structure-guided multi-head attention} module that injects section-level co-occurrence patterns as attention bias, allowing the model to better capture dependencies among clinical note sections.
        
        \item Introduced \textbf{co-missingness-aware feature modeling} and \textbf{disentangled representation learning} to distinguish semantic context from missingness-related signals and improve robustness under incomplete clinical features.
        
        \item Evaluated the framework across multiple missing-feature settings and downstream prediction tasks, with ablation studies validating the effectiveness of co-occurrence-guided attention, co-missingness modeling, and disentanglement components.
        
    \end{highlights}
\end{onecolentry}

\vspace{0.10 cm}

\begin{twocolentry}{
    Big Data 2024
}
    \textbf{Context-Specific Feature Augmentation for Social Determinants of Health Extraction}
\end{twocolentry}

\vspace{0.05 cm}
\begin{onecolentry}
    \begin{highlights}
        \item Developed a \textbf{literature-guided retrieval augmentation} framework for \textbf{social determinants of health extraction} from clinical narratives, enriching clinical text representations with external scientific knowledge.

        \item Employed an entity-aware retrieval module that retrieves semantically relevant scientific articles for each clinical narrative using entity-level matching and semantic similarity.

        \item Designed a \textbf{context-aware feature fusion} module with a \textbf{gated mechanism} and \textbf{dual residual connections} to adaptively integrate literature-derived features with clinical note representations, improving SDOH evidence modeling and extraction performance.


        \item Conducted empirical evaluations on SDOH extraction tasks, analyzing how the number of retrieved scientific articles affects extraction performance and performing detailed case studies to examine the association between retrieved literature content and downstream SDOH labels, improving the interpretability of model decisions.

    \end{highlights}
\end{onecolentry}


\begin{twocolentry}{
    2024
}
    \textbf{Boosting Social Determinants of Health Extraction with Semantic Knowledge Augmented Large Language Model}
\end{twocolentry}

\vspace{0.05 cm}
\begin{onecolentry}
    \begin{highlights}
        \item Developed a \textbf{semantic knowledge-augmented LLM} framework for extracting social determinants of health (SDoH) from unstructured electronic health records, addressing data scarcity and class imbalance.

        \item Designed a \textbf{UMLS-guided data augmentation} strategy that leverages MetaMap to identify biomedical concepts and incorporates semantic types into clinical narratives to enrich model representations.

        \item Applied \textbf{FLAN-T5} models at multiple scales (Base, Large, and XL) to multi-label SDoH classification, investigating the effectiveness of semantic knowledge integration for underrepresented categories.

        \item Evaluated the framework on MIMIC-III SDoH data, improving the FLAN-T5 XL macro-F1 from \textbf{0.8423 to 0.8824}, with ablation studies examining the contribution of different semantic knowledge categories.

    \end{highlights}
\end{onecolentry}


\begin{twocolentry}{
    2026 -- Present
}
    \textbf{SCoPe: Semantic Context-Adaptive Prototypical Learning for Clinical Outcome Prediction}
\end{twocolentry}

\vspace{0.05 cm}
\begin{onecolentry}
    \begin{highlights}
        \item Developed \textbf{SCoPe}, a novel framework for clinical outcome prediction that adaptively constructs variable-length semantic units from long clinical documents, overcoming the limitations of fixed-boundary text segmentation.

        \item Designed a \textbf{semantic context-adaptive aggregation} module that learns token-level semantic boundaries through contextual similarity and merges adjacent tokens into coherent phrase-level representations.

        \item Introduced \textbf{class-conditional prototypical learning} with phrase-level attention to capture discriminative clinical evidence and provide interpretable representations for outcome prediction.

        \item Proposed \textbf{distributional alignment regularization} using KL divergence to stabilize prototype learning under severe class imbalance, improving robustness on challenging clinical prediction tasks.

        \item Evaluated SCoPe on MIMIC-III, MIMIC-IV, and Hyperpartisan benchmarks, improving mortality prediction F1 from \textbf{0.268 to 0.308} over Clinical-ModernBERT. 
    \end{highlights}
\end{onecolentry}


\begin{twocolentry}{
    2026 -- Present
}
    \textbf{OPD-Aha: From Linguistic Momentum to Visual Reflection in Multimodal On-Policy Distillation}
\end{twocolentry}

\vspace{0.05 cm}
\begin{onecolentry}
    \begin{highlights}
        \item Collaborated on \textbf{multimodal on-policy distillation} research to improve vision-language model reasoning by mitigating the influence of erroneous student-generated prefixes on privileged teacher supervision.

        \item Contributed to algorithmic discussions on \textbf{visual preference-based target reconstruction}, which contrasts teacher predictions under privileged visual evidence and visual-null inputs to recover corrective supervision.

        \item Participated in \textbf{post-training Qwen3.5-4B and 9B vision-language models} using teacher--student distillation with privileged visual information and reconstructed token-level supervision.

        \item Contributed to experimental validation and \textbf{benchmark evaluation} across fine-grained perception, high-resolution visual understanding, and multimodal reasoning tasks.

        \item The proposed framework improved average accuracy over Vision-OPD by \textbf{4.0 and 3.2 percentage points} on 4B and 9B models, respectively.
    \end{highlights}
